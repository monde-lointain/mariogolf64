extern u16 D_800BE6E0[];

void func_8004D4B8(char* str, s32 count, u16* out) {
    s32 row = 0;

    if (count > 0) {
        do {
            s32 c = *str;
            s32 t1;
            s32 t0;
            s32 t2;

            if (c == 0) {
                return;
            }
            if (c < 0x20) {
                c = 0x20;
            }
            if (c > 0x7A) {
                c = 0x20;
            }
            c -= 0x20;
            t1 = ((c & 0x1F) << 1) | ((c / 32) << 9);
            t2 = 0;
            t0 = row;
            do {
                s32 base = t1;
                s32 a3;

                for (a3 = 0; a3 < 2; a3++) {
                    out[t0 * 2 + a3] = D_800BE6E0[base + a3];
                }
                t1 += 0x40;
                t2++;
                t0 += count;
            } while (t2 < 8);
            row++;
            str++;
        } while (row < count);
    }
}
