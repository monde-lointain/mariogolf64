/* Symbols that splat names in the asm but leaves for the linker to define */

/* rspboot ends where the n_audio microcode begins, an address splat already
   gives to n_aspMainTextStart */
rspbootTextEnd = 0x800B3F20;

/* Called while the game is halted for debugging. The retail build
   doesn't link it: its calls have a jump target of 0, which is
   0x80000000 for code in KSEG0. */
EnterDevelopmentDebugger = 0x80000000;

/* Overlay 0 reads the z of g_SimulatedShot's end position under this name,
   which splat keeps even though the struct covers the address */
D_8024DD34_ovl_04 = g_SimulatedShot + 0x14;

/* The sound pointer bank ends where the wave bank begins, an address
   splat already gives to g_SoundWaveBankRomStart */
g_SoundPointerBankRomEnd = 0x18BE00;
