/* Symbols that the objects use but none defines */

/* The variables in low memory that IPL3 sets, and the buffer there that
   keeps its contents through a reset */
osTvType = 0x80000300;
osRomBase = 0x80000308;
osResetType = 0x8000030C;
osAppNMIBuffer = 0x8000031C;

/* rspboot ends where the n_audio microcode begins */
rspbootTextEnd = n_aspMainTextStart;

/* Called while the game is halted for debugging. The retail build
   doesn't link it: its calls have a jump target of 0, which is
   0x80000000 for code in KSEG0. */
EnterDevelopmentDebugger = 0x80000000;

/* DrawPressingHand reads this g_GameBoyTransfer field as a scalar */
g_PressingHandImage = g_GameBoyTransfer + 0x40;

/* The files that the game reads from ROM, under the names its code uses */
g_SoundPointerBankRomStart = sound_pointer_bank_ROM_START;
g_SoundPointerBankRomEnd = sound_pointer_bank_ROM_END;
g_SoundWaveBankRomStart = sound_wave_bank_ROM_START;
g_SoundEffectBankRomStart = sound_effect_bank_ROM_START;
g_SoundEffectBankRomEnd = sound_effect_bank_ROM_END;
g_AssetArchiveRomStart = asset_archive_ROM_START;
g_GlyphFileRomStart = text_glyphs_ROM_START;
g_MessageFileRomStart = text_messages_ROM_START;
g_StaffTextRomStart = text_staff_roll_ROM_START;

/* The hole file path formats, which BuildHole has as string literals and
   the asm of SaveHoleFiles reads under these names */
f_VerticesPathFormat = 0x80255168;
f_MidpointsPathFormat = 0x80255188;
f_TreesPathFormat = 0x802551A8;
f_Unk6PathFormat = 0x802551C8;
f_PointsPathFormat = 0x802551E8;
f_TrianglesPathFormat = 0x80255208;
f_AttributesPathFormat = 0x80255228;

/* Main draws from vertex buffers at a fixed place in the Expansion Pak's
   memory, which the hole editor's memory must end before */
